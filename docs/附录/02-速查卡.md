## 3. 定义工具

见 [3.2 模型调用与工具](../03-LangChain实战/02-模型调用与工具.md)

```python
from pydantic import BaseModel, Field

from langchain.tools import tool


# ① 最简：docstring 就是给模型的"何时使用"说明书
@tool
def search_knowledge_base(query: str, k: int = 3) -> str:
    """搜索内部知识库，返回与查询最相关的资料片段。

    Args:
        query: 检索关键词。
        k: 返回片段数量，默认 3。
    """
    return f"关于「{query}」的前 {k} 条结果"


# ② 多参数：用 args_schema 做参数校验（推荐）
class WeatherInput(BaseModel):
    """查询天气的输入参数。"""

    city: str = Field(description="城市名称，如 北京、上海")
    date: str = Field(default="today", description="日期 YYYY-MM-DD，默认今天")


@tool(args_schema=WeatherInput)
def get_weather(city: str, date: str = "today") -> str:
    """查询指定城市指定日期的天气。"""
    return f"{city} {date}：晴，25°C"
```

## 4. 创建 Agent（`create_agent`）

见 [3.1 LangChain v1 核心组件](../03-LangChain实战/01-LangChain核心组件.md)

```python
from langchain.agents import create_agent
from langchain.agents.middleware import SummarizationMiddleware
from langgraph.checkpoint.memory import InMemorySaver


# ① 最小：只有模型
agent_min = create_agent(model="openai:gpt-5.5", tools=[])

# ② 带工具 + 系统提示词
agent_tools = create_agent(
    model="openai:gpt-5.5",
    tools=[search_knowledge_base, get_weather],
    system_prompt="你是知识库助手，需要资料时先检索，再依据结果回答。",
)

# ③ 带 checkpointer（多轮记忆）
agent_mem = create_agent(
    model="openai:gpt-5.5",
    tools=[search_knowledge_base],
    checkpointer=InMemorySaver(),        # 生产换 PostgresSaver
)

# ④ 带中间件（接近阈值自动压缩历史）
agent_mw = create_agent(
    model="openai:gpt-5.5",
    tools=[search_knowledge_base],
    middleware=[
        SummarizationMiddleware(
            model="openai:gpt-5.5",
            trigger={"tokens": 120_000},   # 达到阈值触发压缩
            keep={"tokens": 20_000},       # 最近 2 万 token 保留原文
        ),
    ],
)
```

## 5. 调用与流式

见 [3.4 构建聊天 Agent](../03-LangChain实战/04-构建聊天Agent.md)

```python
from langchain.agents import create_agent
from langgraph.checkpoint.memory import InMemorySaver

agent = create_agent(
    model="openai:gpt-5.5",
    tools=[search_knowledge_base],
    checkpointer=InMemorySaver(),
)

config = {"configurable": {"thread_id": "s1"}}

# ① 一次性调用
result = agent.invoke({"messages": [{"role": "user", "content": "你好"}]}, config)
print(result["messages"][-1].content)

# ② 流式（messages 模式：逐 token；适合前端打字机效果）
for message_chunk, metadata in agent.stream(
    {"messages": [{"role": "user", "content": "讲讲 RAG"}]},
    config,
    stream_mode="messages",
):
    if message_chunk.content:
        print(message_chunk.content, end="", flush=True)

# ③ updates 模式：逐节点更新（观察 Agent 每一步）
for step in agent.stream(
    {"messages": [{"role": "user", "content": "算一下 128*256"}]},
    config,
    stream_mode="updates",
):
    print(step)

# ④ 异步版本
import asyncio

async def main() -> None:
    async for chunk in agent.astream(
        {"messages": [{"role": "user", "content": "你好"}]}, config, stream_mode="messages"
    ):
        print(chunk[0].content, end="", flush=True)

asyncio.run(main())
```

## 6. 结构化输出

见 [3.2 模型调用与工具](../03-LangChain实战/02-模型调用与工具.md)

```python
from pydantic import BaseModel, Field

from langchain.agents import create_agent
from langchain.agents.structured_output import ProviderStrategy, ToolStrategy


class WeatherReport(BaseModel):
    """天气报告。"""

    city: str = Field(description="城市名称")
    temperature: float = Field(description="温度（摄氏度）")
    condition: str = Field(description="天气状况")


# ① 直接传 Pydantic 模型（AutoStrategy，自动选最优策略）
agent = create_agent(model="openai:gpt-5.5", tools=[], response_format=WeatherReport)
# ② 厂商原生结构化输出：ProviderStrategy(WeatherReport)
# ③ 兼容所有支持 function calling 的模型：ToolStrategy(WeatherReport)

result = agent.invoke({"messages": [{"role": "user", "content": "杭州今天晴天，25 度"}]})
report: WeatherReport = result["structured_response"]     # 已验证的 Pydantic 实例
print(report.city, report.temperature, report.condition)
```

## 7. 记忆与多会话（`thread_id`）

见 [2.4 记忆机制](../02-核心概念/04-记忆机制.md)

```python
from langchain.agents import create_agent
from langgraph.checkpoint.memory import InMemorySaver

agent = create_agent(
    model="openai:gpt-5.5",
    tools=[],
    system_prompt="你是友好的中文助手。",
    checkpointer=InMemorySaver(),
)


def chat(thread_id: str, text: str) -> str:
    """同一个 thread_id = 同一会话；不同 thread_id 完全隔离。"""
    config = {"configurable": {"thread_id": thread_id}}
    result = agent.invoke({"messages": [{"role": "user", "content": text}]}, config)
    return result["messages"][-1].content


print(chat("user-1", "我叫小明"))
print(chat("user-1", "我叫什么名字？"))   # → 你叫小明
print(chat("user-2", "我叫什么名字？"))   # → 不知道（会话隔离）
```

## 8. LangGraph（状态图 + reducer + 持久化 + 中断）

见 [4.1 LangGraph 基础](../04-LangGraph进阶/01-LangGraph基础.md) · [4.2 状态管理与节点](../04-LangGraph进阶/02-状态管理与节点.md) · [4.5 实战工作流 Agent](../04-LangGraph进阶/05-实战工作流Agent.md)

```python
import operator
from typing import Annotated, TypedDict

from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt


# ① 状态：TypedDict + reducer
class GraphState(TypedDict):
    topic: str
    steps: Annotated[list[str], operator.add]   # 追加，不覆盖
    draft: str
    approved: bool


# ② 节点：接收状态，返回"部分更新"
def plan_node(state: GraphState) -> dict:
    return {"steps": ["plan"], "draft": f"关于「{state['topic']}」的草稿"}


def approval_node(state: GraphState) -> dict:
    """中断等待人工审批（HITL）"""
    decision = interrupt({"question": "是否批准发布？", "draft": state["draft"]})
    return {"approved": bool(decision.get("approved"))}


def finalize_node(state: GraphState) -> dict:
    return {"steps": ["finalize"]}


def route_after_approval(state: GraphState) -> str:
    return "finalize" if state["approved"] else END


# ③ 构图 + 编译（带 checkpointer）
builder = StateGraph(GraphState)
builder.add_node("plan", plan_node)
builder.add_node("approval", approval_node)
builder.add_node("finalize", finalize_node)
builder.add_edge(START, "plan")
builder.add_edge("plan", "approval")
builder.add_conditional_edges("approval", route_after_approval, {"finalize": "finalize", END: END})
builder.add_edge("finalize", END)

app = builder.compile(checkpointer=InMemorySaver())
config = {"configurable": {"thread_id": "wf-1"}}

# ④ 执行：第一次会停在 approval 的 interrupt 处
result = app.invoke({"topic": "AI Agent", "steps": [], "draft": "", "approved": False}, config)
if result.get("__interrupt__"):
    # ⑤ 用同一 thread_id + Command(resume=...) 恢复
    result = app.invoke(Command(resume={"approved": True}), config)

print(result["steps"])   # ['plan', 'finalize']
```

## 9. RAG 最小闭环（加载 → 切分 → Chroma → 检索）

见 [5.1 RAG 原理](../05-RAG系统/01-RAG原理.md) · [3.3 向量存储与检索](../03-LangChain实战/03-向量存储与检索.md)

```python
import os

from langchain_chroma import Chroma
from langchain_core.documents import Document
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.runnables import RunnablePassthrough
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter


def get_llm() -> ChatOpenAI:
    return ChatOpenAI(
        model=os.getenv("LLM_MODEL_NAME", "gpt-5.5"),
        api_key=os.getenv("LLM_API_KEY"),
        base_url=os.getenv("LLM_BASE_URL"),
        temperature=0,
    )


# ① 加载（示例直接构造；真实项目用 TextLoader / PyPDFLoader 等）
raw_text = "RAG 通过检索增强生成，缓解幻觉。\n\nChroma 是一个轻量向量库，写入即落盘。"

# ② 切分
splitter = RecursiveCharacterTextSplitter(chunk_size=300, chunk_overlap=50)
docs = [Document(page_content=chunk, metadata={"source": "intro.md"})
        for chunk in splitter.split_text(raw_text)]

# ③ 存入 Chroma（写入即落盘，禁止调用 vectorstore.persist()）
embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
vectorstore = Chroma.from_documents(
    documents=docs,
    embedding=embeddings,
    collection_name="cheatsheet_rag",
    persist_directory="./chroma_db",
)

# ④ 检索
retriever = vectorstore.as_retriever(search_kwargs={"k": 2})


def format_docs(hits: list[Document]) -> str:
    return "\n\n".join(f"[{d.metadata.get('source', '?')}] {d.page_content}" for d in hits)


prompt = ChatPromptTemplate.from_template(
    "只依据参考信息回答，并标注来源。\n\n参考信息：\n{context}\n\n问题：{question}"
)

rag_chain = (
    {"context": retriever | format_docs, "question": RunnablePassthrough()}
    | prompt
    | get_llm()
    | StrOutputParser()
)

print(rag_chain.invoke("Chroma 有什么特点？"))
```

## 10. FastAPI 流式 SSE 骨架

见 [7.1 API 服务部署](../07-生产部署/01-API服务部署.md)

```python
# main.py —— 运行：uv run uvicorn main:app --reload
from collections.abc import AsyncIterator

from fastapi import FastAPI, Header, HTTPException
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import StreamingResponse
from pydantic import BaseModel, Field

from langchain.agents import create_agent
from langgraph.checkpoint.memory import InMemorySaver

app = FastAPI(title="Agent API")

# CORS：生产环境请改为白名单，勿用 "*"
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

agent = create_agent(
    model="openai:gpt-5.5",
    tools=[],
    system_prompt="你是一个简洁的中文助手。",
    checkpointer=InMemorySaver(),
)


class ChatRequest(BaseModel):
    message: str = Field(..., min_length=1, max_length=4000)
    session_id: str = Field(default="default")


async def event_stream(text: str, session_id: str) -> AsyncIterator[str]:
    """把 Agent 的 messages 流式输出转成 SSE 帧。"""
    config = {"configurable": {"thread_id": session_id}}
    async for chunk, _meta in agent.astream(
        {"messages": [{"role": "user", "content": text}]},
        config,
        stream_mode="messages",
    ):
        if chunk.content:
            # 关键：SSE 每帧以 data: 开头，以空行结束
            yield f"data: {chunk.content}\n\n"
    yield "data: [DONE]\n\n"


@app.post("/chat/stream")
async def chat_stream(req: ChatRequest, x_api_key: str = Header(default="")):
    if x_api_key != "change-me":                      # 简易鉴权，生产用正规方案
        raise HTTPException(status_code=401, detail="invalid api key")
    return StreamingResponse(
        event_stream(req.message, req.session_id),
        media_type="text/event-stream",
        headers={"Cache-Control": "no-cache", "X-Accel-Buffering": "no"},  # 关闭反代缓冲
    )
```

> ⚠️ **SSE 三个坑**：① 反代（Nginx）默认缓冲会导致"一次性吐出"，需 `X-Accel-Buffering: no`；② 必须设置 `media_type="text/event-stream"`；③ 每帧以 `\n\n` 结尾。
