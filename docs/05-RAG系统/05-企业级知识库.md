## 四、多源文档处理（`document_processor.py`）

统一把不同来源、不同格式的文档，转成**带完整元数据**的 `Document` 分块。

```python
"""多源文档处理：加载 → 清洗 → 分块 → 元数据"""
import hashlib
import re
from pathlib import Path

from langchain_community.document_loaders import (
    DirectoryLoader, TextLoader, PyPDFLoader, Docx2txtLoader,
)
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_core.documents import Document

from config import KB_CONFIG


class DocumentProcessor:
    """多格式文档处理器"""

    LOADERS = {
        ".txt": (TextLoader, {"encoding": "utf-8"}),
        ".md": (TextLoader, {"encoding": "utf-8"}),
        ".pdf": (PyPDFLoader, {}),
        ".docx": (Docx2txtLoader, {}),
    }

    def __init__(self, default_department: str = "public"):
        self.default_department = default_department

    # ---------- 加载 ----------
    def load(self, path: str) -> list[Document]:
        p = Path(path)
        docs: list[Document] = []
        if p.is_file():
            loader_cls, kwargs = self.LOADERS[p.suffix.lower()]
            docs = loader_cls(str(p), **kwargs).load()
        else:
            for ext, (loader_cls, kwargs) in self.LOADERS.items():
                try:
                    loader = DirectoryLoader(
                        str(p), glob=f"**/*{ext}", loader_cls=loader_cls,
                        loader_kwargs=kwargs, show_progress=False,
                    )
                    docs.extend(loader.load())
                except Exception as e:  # noqa: BLE001
                    print(f"  ⚠️ 跳过 {ext}: {e}")
        print(f"📂 加载 {len(docs)} 个文档片段")
        return docs

    # ---------- 清洗 ----------
    @staticmethod
    def clean(docs: list[Document]) -> list[Document]:
        for doc in docs:
            text = re.sub(r"[ \t]+", " ", doc.page_content)
            text = re.sub(r"\n{3,}", "\n\n", text)
            doc.page_content = text.strip()
        return docs

    # ---------- 分块 + 元数据 ----------
    def split(
        self,
        docs: list[Document],
        department: str | None = None,
    ) -> list[Document]:
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=KB_CONFIG["chunk_size"],
            chunk_overlap=KB_CONFIG["chunk_overlap"],
            separators=["\n\n", "\n", "。", "！", "？", ".", " ", ""],
        )
        chunks = splitter.split_documents(docs)
        dept = department or self.default_department
        for i, chunk in enumerate(chunks):
            source = Path(chunk.metadata.get("source", "unknown")).name
            # 稳定 id：来源 + 内容指纹 → 重复写入同 id 即覆盖（增量更新关键）
            fingerprint = hashlib.md5(
                f"{source}::{chunk.page_content}".encode("utf-8")
            ).hexdigest()
            chunk.metadata.update({
                "source": source,
                "department": dept,          # 权限过滤依据
                "chunk_index": i,
                "chunk_id": fingerprint,
            })
        print(f"✂️ 分割为 {len(chunks)} 个块（department={dept}）")
        return chunks

    def process(self, path: str, department: str | None = None) -> list[Document]:
        return self.split(self.clean(self.load(path)), department=department)
```

## 五、知识库：增量更新 + 权限过滤（`knowledge_base.py`）

两个生产要点：

1. **增量更新**：用**稳定 id**（来源 + 内容指纹）写入，未变内容不重复；用 `manifest.json` 记录每个来源的块 id，重新索引时**删除已失效的块**。
2. **权限过滤**：分块元数据带 `department`，检索时用 `filter` 只召回用户有权访问的部门。

```python
"""企业知识库：Chroma + 增量更新 + 权限过滤"""
import json
import os
from typing import List

from langchain_chroma import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain_core.documents import Document

from config import KB_CONFIG, EMBEDDING_MODEL, MANIFEST_PATH


class KnowledgeBase:
    """企业知识库"""

    def __init__(self):
        self.embeddings = OpenAIEmbeddings(model=EMBEDDING_MODEL)
        # 指定 persist_directory 后写入即自动落盘（禁止调用已弃用的 persist()）
        self.vectorstore = Chroma(
            collection_name=KB_CONFIG["collection_name"],
            embedding_function=self.embeddings,
            persist_directory=KB_CONFIG["persist_directory"],
        )
        self.manifest: dict[str, list[str]] = self._load_manifest()

    # ---------- 增量更新清单 ----------
    def _load_manifest(self) -> dict[str, list[str]]:
        if os.path.exists(MANIFEST_PATH):
            with open(MANIFEST_PATH, encoding="utf-8") as f:
                return json.load(f)
        return {}

    def _save_manifest(self) -> None:
        with open(MANIFEST_PATH, "w", encoding="utf-8") as f:
            json.dump(self.manifest, f, ensure_ascii=False, indent=2)

    # ---------- 写入（增量） ----------
    def upsert(self, chunks: List[Document]) -> dict:
        """按稳定 id 写入；同一来源只保留最新内容"""
        by_source: dict[str, list[Document]] = {}
        for chunk in chunks:
            by_source.setdefault(chunk.metadata["source"], []).append(chunk)

        stats = {"added": 0, "removed": 0}
        for source, source_chunks in by_source.items():
            new_ids = [c.metadata["chunk_id"] for c in source_chunks]
            old_ids = set(self.manifest.get(source, []))

            # 1) 写入/覆盖（langchain-chroma 对同 id 执行 upsert）
            self.vectorstore.add_documents(source_chunks, ids=new_ids)

            # 2) 删除该来源已失效的块（内容被改/删）
            stale = list(old_ids - set(new_ids))
            if stale:
                self.vectorstore.delete(stale)
                stats["removed"] += len(stale)

            self.manifest[source] = new_ids
            stats["added"] += len(new_ids)

        self._save_manifest()
        print(f"✅ 更新完成：写入 {stats['added']}，清理 {stats['removed']}")
        return stats

    def remove_source(self, source: str) -> None:
        """整源删除（文档下线）"""
        ids = self.manifest.pop(source, [])
        if ids:
            self.vectorstore.delete(ids)
            self._save_manifest()
        print(f"🗑️ 已删除来源 {source}（{len(ids)} 块）")

    # ---------- 权限过滤检索 ----------
    def search(
        self,
        query: str,
        user_roles: list[str],
        k: int | None = None,
    ) -> List[Document]:
        """只检索用户有权访问的部门文档"""
        k = k or KB_CONFIG["retriever_k"]
        return self.vectorstore.similarity_search(
            query,
            k=k,
            filter={"department": {"$in": user_roles}},   # 权限过滤
        )

    def stats(self) -> dict:
        return {
            "total_chunks": self.vectorstore._collection.count(),
            "sources": len(self.manifest),
            "persist_directory": KB_CONFIG["persist_directory"],
        }
```

> ⚠️ **权限过滤是"检索层"兜底**。真实系统应在**服务端鉴权**（谁能访问哪些 `department`）后，再把允许的部门列表传给 `search()`，切勿信任前端传入的角色。

## 六、检索优化（`retriever.py`）

复用 [5.4 检索优化技巧](./04-优化策略.md) 的思路：多查询 + 重排。这里给出与知识库对接的薄封装。

```python
"""检索优化：多查询 + 重排（对接 KnowledgeBase 的权限过滤）"""
from typing import List

from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from langchain_openai import ChatOpenAI

from config import LLM_CONFIG


class OptimizedRetriever:
    """带权限过滤的多查询 + 重排检索器"""

    def __init__(self, knowledge_base):
        self.kb = knowledge_base
        self.llm = ChatOpenAI(**LLM_CONFIG)
        self._expand_prompt = ChatPromptTemplate.from_template(
            "生成 2 条与下面查询同义但措辞不同的检索查询，每行一条，不要编号。\n\n查询：{query}"
        )

    def _expand(self, query: str) -> list[str]:
        text = (self._expand_prompt | self.llm | StrOutputParser()).invoke({"query": query})
        variants = [q.strip("-• \t") for q in text.splitlines() if q.strip()]
        return [query] + variants[:2]

    def retrieve(self, query: str, user_roles: list[str], k: int = 4) -> List[Document]:
        # 1. 多查询 + 权限过滤召回
        all_docs: list[Document] = []
        seen: set[str] = set()
        for q in self._expand(query):
            for doc in self.kb.search(q, user_roles=user_roles, k=k):
                key = doc.metadata.get("chunk_id", doc.page_content[:80])
                if key not in seen:
                    seen.add(key)
                    all_docs.append(doc)

        # 2. LLM 重排（生产可换 CrossEncoder / Cohere Rerank）
        scored: list[tuple[float, Document]] = []
        for doc in all_docs:
            prompt = (
                "评估文档与问题的相关性，只输出 0-10 的数字。\n"
                f"问题：{query}\n文档：{doc.page_content[:300]}\n分数："
            )
            try:
                score = float(self.llm.invoke(prompt).content.strip())
            except ValueError:
                score = 5.0
            scored.append((score, doc))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [doc for _, doc in scored[:k]]
```

## 七、QA 问答链（`qa_chain.py`）

```python
"""QA 问答链：检索 → 增强 → 生成，并附来源"""
from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

from config import LLM_CONFIG


class QAChain:
    """问答链"""

    def __init__(self, retriever):
        self.retriever = retriever
        self.llm = ChatOpenAI(**LLM_CONFIG)
        self.prompt = ChatPromptTemplate.from_messages([
            ("system",
             "你是企业知识库问答助手。只依据参考信息回答；"
             "若参考信息不足，回答'未找到相关信息'；回答时标注来源文件名。"),
            ("human", "参考信息：\n{context}\n\n问题：{question}"),
        ])
        self.chain = self.prompt | self.llm | StrOutputParser()

    def answer(self, question: str, user_roles: list[str]) -> dict:
        docs = self.retriever.retrieve(question, user_roles=user_roles)
        context = "\n\n".join(
            f"[来源 {i + 1}: {d.metadata.get('source', '未知')}]\n{d.page_content}"
            for i, d in enumerate(docs)
        )
        answer = self.chain.invoke({"context": context, "question": question})
        return {
            "question": question,
            "answer": answer,
            "sources": [
                {"file": d.metadata.get("source", "未知"),
                 "department": d.metadata.get("department", "public")}
                for d in docs
            ],
        }
```

## 八、主入口（`main.py`）

```python
"""企业知识库主程序"""
from document_processor import DocumentProcessor
from knowledge_base import KnowledgeBase
from retriever import OptimizedRetriever
from qa_chain import QAChain


class EnterpriseKB:
    """企业知识库系统"""

    def __init__(self):
        print("🚀 初始化企业知识库系统...")
        self.processor = DocumentProcessor()
        self.kb = KnowledgeBase()
        self.retriever = OptimizedRetriever(self.kb)
        self.qa_chain = QAChain(self.retriever)

    def ingest(self, path: str, department: str = "public") -> None:
        """接入一个来源（增量更新）"""
        chunks = self.processor.process(path, department=department)
        self.kb.upsert(chunks)
        print(f"📊 知识库状态: {self.kb.stats()}")

    def ask(self, question: str, user_roles: list[str]) -> dict:
        return self.qa_chain.answer(question, user_roles=user_roles)


def main():
    kb = EnterpriseKB()

    # 1) 多源接入：不同目录 → 不同部门（权限）
    kb.ingest("./documents/public", department="public")
    kb.ingest("./documents/hr", department="hr")

    # 2) 普通员工只能看到 public
    result = kb.ask("公司的远程办公政策是什么？", user_roles=["public"])
    print(f"\n🤖 {result['answer']}")
    print("📚 来源:", [s["file"] for s in result["sources"]])

    # 3) HR 可额外看到 hr 部门文档
    result = kb.ask("年假有多少天？", user_roles=["public", "hr"])
    print(f"\n🤖 {result['answer']}")
    print("📚 来源:", [s["file"] for s in result["sources"]])


if __name__ == "__main__":
    main()
```

## 九、增量更新的三种情形

| 情形 | 触发 | 行为 |
|------|------|------|
| **新增文档** | 新文件进入目录 | 生成新 id 并写入 |
| **修改文档** | 内容变化 → 指纹变化 | 新 id 写入 + 旧 id 删除 |
| **删除文档** | 文件被移除 | 调 `remove_source(source)` 清理 |

> 💡 由于 id 由"来源 + 内容指纹"生成，**未改动的块 id 不变**——重复执行 `ingest` 不会产生冗余，真正实现了"只更新变化的"。
