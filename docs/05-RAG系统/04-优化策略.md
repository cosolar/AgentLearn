> **RAG 系统的瓶颈往往不在 LLM，而在检索环节。检索不到？回答必然出错。**

基础 RAG（文档分块 → 向量化 → 检索 → 生成）在真实场景会碰到各种问题：检索不相关、遗漏关键信息、LLM 被噪声干扰。本章系统讲解**查询改写、HyDE、混合检索（含可用 BM25 实现）、重排序、上下文压缩、评估**，让你的 RAG 从"能用"变"好用"。

## 二、常见检索问题

| 问题 | 表现 | 原因 |
|------|------|------|
| **检索不相关** | 结果和问题没关系 | 查询词不准确 |
| **遗漏关键信息** | 遗漏了最重要的文档 | 纯语义检索漏掉关键词精确匹配 |
| **上下文不完整** | 结果碎片化，缺上下文 | chunk 过小 |
| **信息冗余** | 重复的相似结果 | 未使用 MMR |
| **低分高质** | 低分文档其实准确 | 语义偏移 |

## 三、查询侧优化

### 3.1 查询改写（Query Rewriting）

把口语化、含糊的查询改写为**检索友好**的表达。

```python
import os

from langchain_openai import ChatOpenAI
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser


def get_llm(temperature: float = 0.0) -> ChatOpenAI:
    return ChatOpenAI(
        model=os.getenv("LLM_MODEL_NAME", "gpt-5.5"),
        api_key=os.getenv("LLM_API_KEY"),
        base_url=os.getenv("LLM_BASE_URL"),
        temperature=temperature,
    )


class QueryRewriter:
    """查询改写：消歧、补全、去口语化"""

    def __init__(self):
        prompt = ChatPromptTemplate.from_template(
            "把下面的用户问题改写为一条更适合检索的查询语句（保留原意，去掉口语，"
            "补全隐含主体）。只输出改写后的查询，不要解释。\n\n原始问题：{query}"
        )
        self.chain = prompt | get_llm() | StrOutputParser()

    def rewrite(self, query: str) -> str:
        return self.chain.invoke({"query": query}).strip()
```

### 3.2 多查询（Multi-Query）

一条问题 → 多个角度的查询 → 分别检索后合并，提升**召回**。

```python
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser


class MultiQueryGenerator:
    """把一个查询扩展为多个变体"""

    def __init__(self, n: int = 3):
        self.n = n
        prompt = ChatPromptTemplate.from_template(
            "针对下面的原始查询，生成 {n} 个不同角度、不同措辞的检索查询。"
            "每行一条，不要编号，不要解释。\n\n原始查询：{query}"
        )
        self.chain = prompt | get_llm(temperature=0.3) | StrOutputParser()

    def generate(self, query: str) -> list[str]:
        text = self.chain.invoke({"query": query, "n": self.n})
        variants = [q.strip("-• \t") for q in text.splitlines() if q.strip()]
        # 始终包含原始查询
        return [query] + variants[: self.n]
```

### 3.3 HyDE（假设文档嵌入）

先让 LLM 生成一段**假设答案**，再用它去检索。因为"答案"与"文档"的语义分布更接近，检索精度往往更高。

```python
class HyDERetriever:
    """HyDE：先生成假设答案，再检索"""

    def __init__(self, retriever, k: int = 3):
        self.retriever = retriever
        self.k = k
        prompt = ChatPromptTemplate.from_template(
            "针对下面的问题，写一段详细的假设性答案（像百科条目一样，包含相关事实与概念）。"
            "即使不确定也要写得合理。只输出这段答案。\n\n问题：{query}"
        )
        self.chain = prompt | get_llm(temperature=0.3) | StrOutputParser()

    def retrieve(self, query: str):
        hypothetical = self.chain.invoke({"query": query})
        return self.retriever.invoke(hypothetical)[: self.k]


# 为什么有效？
# 问题："装饰器是干什么的？"（很短，语义稀薄）
#   ↓ 生成假设答案
# "Python 装饰器是一种用于修改函数行为的设计模式，通过 @ 语法应用……"（语义丰富）
#   ↓ 用假设答案检索 → 更容易命中真正的文档
```

## 四、混合检索（Hybrid Search）

结合**关键词检索（BM25）**的精确匹配与**向量检索**的语义匹配，互补短板。

```text
用户查询
   │
   ├── 稠密检索（向量）──┐
   │                     ├── 融合（RRF）── 重排（Rerank）── Top-K
   └── 稀疏检索（BM25）──┘
```

### 4.1 可用的 BM25 实现（无额外依赖）

```python
"""混合检索：BM25（自实现）+ 向量 + RRF 融合"""
import math
import re
from typing import List

from langchain_core.documents import Document


class HybridRetriever:
    """关键词 + 向量混合检索，纯 Python 实现 BM25，无需额外依赖"""

    def __init__(self, vectorstore):
        self.vectorstore = vectorstore

    # ---------- 工具方法 ----------
    @staticmethod
    def _tokenize(text: str) -> List[str]:
        """中英文混合分词：英文/数字按词切，中文按单字切"""
        return re.findall(r"[a-zA-Z0-9]+|[\u4e00-\u9fff]", text.lower())

    def _corpus(self) -> List[Document]:
        """从向量库底层集合取出全部文档作为语料"""
        collection = self.vectorstore._collection
        data = collection.get(include=["documents", "metadatas"])
        return [
            Document(page_content=text, metadata=meta or {})
            for text, meta in zip(data["documents"], data["metadatas"])
        ]

    # ---------- BM25 ----------
    def keyword_search(self, query: str, k: int = 3) -> List[Document]:
        corpus = self._corpus()
        if not corpus:
            return []

        tokenized = [self._tokenize(d.page_content) for d in corpus]
        query_tokens = self._tokenize(query)
        if not query_tokens:
            return []

        n = len(tokenized)
        avg_len = sum(len(t) for t in tokenized) / n

        # 文档频率 DF
        df: dict[str, int] = {}
        for tokens in tokenized:
            for token in set(tokens):
                df[token] = df.get(token, 0) + 1

        k1, b = 1.5, 0.75
        scored: list[tuple[float, Document]] = []
        for tokens, doc in zip(tokenized, corpus):
            tf: dict[str, int] = {}
            for token in tokens:
                tf[token] = tf.get(token, 0) + 1

            score = 0.0
            for token in query_tokens:
                if token not in tf:
                    continue
                idf = math.log(1 + (n - df[token] + 0.5) / (df[token] + 0.5))
                denom = tf[token] + k1 * (1 - b + b * len(tokens) / avg_len)
                score += idf * tf[token] * (k1 + 1) / denom
            scored.append((score, doc))

        scored.sort(key=lambda x: x[0], reverse=True)
        return [doc for score, doc in scored[:k] if score > 0]

    # ---------- 向量 ----------
    def vector_search(self, query: str, k: int = 3) -> List[Document]:
        return self.vectorstore.similarity_search(query, k=k)

    # ---------- RRF 融合 ----------
    @staticmethod
    def _rrf_fuse(result_lists: List[List[Document]], k: int = 60) -> List[Document]:
        """Reciprocal Rank Fusion：按排名倒数加权融合多路结果"""
        scores: dict[str, float] = {}
        doc_map: dict[str, Document] = {}
        for results in result_lists:
            for rank, doc in enumerate(results, start=1):
                key = doc.page_content[:120]
                doc_map[key] = doc
                scores[key] = scores.get(key, 0.0) + 1.0 / (k + rank)
        ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True)
        return [doc_map[key] for key, _ in ranked]

    def hybrid_search(self, query: str, k: int = 3, fetch_k: int = 10) -> List[Document]:
        """混合检索：两路各取 fetch_k，RRF 融合后取前 k"""
        keyword_results = self.keyword_search(query, k=fetch_k)
        vector_results = self.vector_search(query, k=fetch_k)
        return self._rrf_fuse([keyword_results, vector_results])[:k]
```

> 💡 语料很大时，`_corpus()` 全量加载会变慢。生产环境建议把 BM25 换成 **Elasticsearch / OpenSearch / Whoosh**，或使用支持稀疏向量的库（如 Qdrant 的 sparse vector）。

## 五、重排序（Re-ranking）

初次检索（召回）追求"不漏"，重排追求"排序准"。用更强的模型对候选重新打分。

### 5.1 交叉编码器重排（本地，推荐）

```python
"""用 sentence-transformers 的 CrossEncoder 做重排（本地模型，无需 API）"""
from typing import List

from sentence_transformers import CrossEncoder
from langchain_core.documents import Document


class CrossEncoderReranker:
    """BGE / MiniLM 交叉编码器重排"""

    def __init__(self, model_name: str = "BAAI/bge-reranker-base"):
        # 首次运行会从 HuggingFace 下载模型
        self.model = CrossEncoder(model_name)

    def rerank(self, query: str, docs: List[Document], top_k: int = 3) -> List[Document]:
        if not docs:
            return []
        pairs = [(query, d.page_content) for d in docs]
        scores = self.model.predict(pairs)
        ranked = sorted(zip(scores, docs), key=lambda x: x[0], reverse=True)
        return [doc for _, doc in ranked[:top_k]]
```

### 5.2 LLM 打分重排（无需额外模型）

```python
import re


class LLMReranker:
    """用 LLM 给每个候选打 0-10 相关性分"""

    def __init__(self):
        self.llm = get_llm()

    def rerank(self, query: str, docs: List[Document], top_k: int = 3) -> List[Document]:
        scored: list[tuple[float, Document]] = []
        for doc in docs:
            prompt = (
                "评估下面文档与问题的相关性，只输出 0-10 的数字。\n"
                f"问题：{query}\n文档：{doc.page_content[:300]}\n分数："
            )
            text = self.llm.invoke(prompt).content
            match = re.search(r"\d+(\.\d+)?", text)
            score = float(match.group()) if match else 5.0
            scored.append((score, doc))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [doc for _, doc in scored[:top_k]]
```

### 5.3 托管重排（生产推荐）

```python
# langchain-cohere：托管 rerank，效果稳定、免运维
# uv add langchain-cohere
from langchain_cohere import CohereRerank

reranker = CohereRerank(model="rerank-v3.5", top_n=3)
reranked_docs = reranker.compress_documents(documents=retrieved_docs, query=query)
```

| 重排方案 | 成本 | 效果 | 适用 |
|----------|------|------|------|
| CrossEncoder（本地） | 低（需算力） | ⭐⭐⭐⭐ | 私有化、无外网 |
| LLM 打分 | 中（token） | ⭐⭐⭐ | 无本地模型时 |
| Cohere / Voyage Rerank | 按量付费 | ⭐⭐⭐⭐⭐ | 生产、追求效果 |

## 六、上下文压缩

检索结果含噪声时，先压缩再喂给 LLM。

```python
class ContextCompressor:
    """只保留与问题相关的信息"""

    def __init__(self):
        self.llm = get_llm()

    def compress(self, query: str, docs: List[Document], max_length: int = 1000) -> str:
        combined = "\n\n".join(d.page_content for d in docs)[:3000]
        prompt = (
            "从下面的文档中提取与问题相关的信息，压缩为简洁上下文，"
            "移除无关内容，保留关键事实、数据与结论。\n\n"
            f"问题：{query}\n\n文档：{combined}\n\n压缩后的上下文："
        )
        return self.llm.invoke(prompt).content[:max_length]
```

## 七、完整优化 RAG

```python
class OptimizedRAG:
    """整合查询改写 + 混合检索 + 重排 + 压缩 + 生成"""

    def __init__(self, vectorstore):
        self.hybrid = HybridRetriever(vectorstore)
        self.rewriter = QueryRewriter()
        self.multi_query = MultiQueryGenerator()
        self.reranker = CrossEncoderReranker()
        self.compressor = ContextCompressor()
        self.llm = get_llm()

    def query(self, question: str) -> dict:
        # 1. 查询改写 + 多查询
        rewritten = self.rewriter.rewrite(question)
        queries = self.multi_query.generate(rewritten)

        # 2. 每个查询做混合检索，再融合
        all_docs: list[Document] = []
        seen: set[str] = set()
        for q in queries:
            for doc in self.hybrid.hybrid_search(q, k=3, fetch_k=8):
                key = doc.page_content[:120]
                if key not in seen:
                    seen.add(key)
                    all_docs.append(doc)

        # 3. 重排
        reranked = self.reranker.rerank(question, all_docs, top_k=4)

        # 4. 压缩
        context = self.compressor.compress(question, reranked)

        # 5. 生成
        answer = self.llm.invoke(
            f"只依据以下信息回答问题，并标注来源。\n\n{context}\n\n问题：{question}\n回答："
        )

        return {
            "question": question,
            "answer": answer.content,
            "retrieved": len(all_docs),
            "used": len(reranked),
            "sources": [d.metadata.get("source", "未知") for d in reranked],
        }
```

## 八、评估指标

| 指标 | 全称 | 衡量什么 | 理想值 |
|------|------|----------|--------|
| **命中率** | Hit Rate | 检索是否包含正确答案 | > 80% |
| **MRR** | Mean Reciprocal Rank | 首个相关结果的排名倒数 | > 0.7 |
| **NDCG** | Normalized DCG | 排序质量 | > 0.8 |
| **精确率** | Precision | 检索结果的相关性 | > 70% |
| **召回率** | Recall | 是否覆盖所有相关文档 | > 80% |

```python
"""检索质量评估：Hit Rate / MRR / NDCG"""
import math


def hit_rate(retrieved, relevant) -> float:
    return float(bool({d.metadata.get("id") for d in retrieved} & relevant))


def mrr(retrieved, relevant) -> float:
    for i, doc in enumerate(retrieved, start=1):
        if doc.metadata.get("id") in relevant:
            return 1.0 / i
    return 0.0


def ndcg(retrieved, relevant, k: int = 5) -> float:
    dcg = 0.0
    for i, doc in enumerate(retrieved[:k], start=1):
        rel = 1.0 if doc.metadata.get("id") in relevant else 0.0
        dcg += rel / math.log2(i + 1)
    ideal = sum(1.0 / math.log2(i + 1) for i in range(1, min(len(relevant), k) + 1))
    return dcg / ideal if ideal else 0.0


def evaluate_retrieval(retriever, test_queries: list, relevant_docs: dict) -> dict:
    """在带标注的测试集上评估检索器"""
    total = {"hit_rate": 0.0, "mrr": 0.0, "ndcg": 0.0}
    for query in test_queries:
        retrieved = retriever.invoke(query)
        relevant = relevant_docs.get(query, set())
        total["hit_rate"] += hit_rate(retrieved, relevant)
        total["mrr"] += mrr(retrieved, relevant)
        total["ndcg"] += ndcg(retrieved, relevant)
    n = len(test_queries) or 1
    return {k: v / n for k, v in total.items()}
```

> 📌 **端到端评估**：检索指标之外，还可用 **RAGAS**（`uv add ragas`）评估回答质量，核心指标包括 `faithfulness`（忠实度）、`answer_relevancy`（相关性）、`context_precision` / `context_recall`。详见 [6.2 评估与优化](../06-高级主题/02-评估与优化.md)。
