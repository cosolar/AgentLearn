# 7.3 可观测性、监控与日志 —— 让 Agent 运行状态一目了然

> 📅 更新至 2026-10 · 技术栈基线见 [教程大纲](../教程大纲.md)

## 📖 导读

> **没有监控的线上系统，就像没有仪表盘的飞机——你飞在天上，却不知道高度、速度和油量。**

Agent 一次请求往往包含 5–15 次 LLM 调用与若干工具调用，**"没有追踪等于在黑盒里调试"**。本章构建完整的可观测体系：**结构化日志 + 指标 + 链路追踪（OpenTelemetry GenAI 语义约定）+ 告警**。

---

## 一、前置知识

| 概念 | 说明 |
|------|------|
| 日志（Logging） | 记录离散事件 |
| 指标（Metrics） | 可聚合的数值（计数、直方图、仪表） |
| 追踪（Tracing） | 一次请求的完整调用链（span 树） |
| 告警（Alerting） | 异常时通知负责人 |
| OTel GenAI | OpenTelemetry 为 LLM/Agent 定义的 `gen_ai.*` 语义约定 |

```text
┌──────────────────────────────────────────────────────────┐
│                        可观测性                            │
├───────────┬───────────────┬───────────────┬──────────────┤

└───────────┴───────────────┴───────────────┴──────────────┘
```

## 三、结构化日志

> ⚠️ **定义顺序很重要**：`JsonFormatter` 必须在 `StructuredLogger` **之前**定义，否则会 `NameError`。

```python
"""结构化日志：JSON 输出，便于采集与检索"""
from __future__ import annotations

import json
import logging
from datetime import datetime, timezone


class JsonFormatter(logging.Formatter):
    """把日志记录序列化为 JSON（必须定义在使用它的类之前）"""

    def format(self, record: logging.LogRecord) -> str:
        payload: dict = {
            "timestamp": datetime.now(timezone.utc).isoformat(),  # 统一 UTC
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
        }
        # 附加通过 extra={"extra_fields": {...}} 传入的自定义字段
        for key, value in getattr(record, "extra_fields", {}).items():
            payload[key] = value
        if record.exc_info:
            payload["exc_info"] = self.formatException(record.exc_info)
        return json.dumps(payload, ensure_ascii=False)


class StructuredLogger:
    """结构化日志封装"""

    def __init__(self, name: str = "agent") -> None:
        self.logger = logging.getLogger(name)
        self.logger.setLevel(logging.INFO)
        if not self.logger.handlers:
            handler = logging.StreamHandler()
            handler.setFormatter(JsonFormatter())
            self.logger.addHandler(handler)
            self.logger.propagate = False

    def _log(self, level: int, event: str, **fields: object) -> None:
        self.logger.log(level, event, extra={"extra_fields": {"event": event, **fields}})

    def log_request(self, session_id: str, message: str) -> None:
        self._log(logging.INFO, "request", session_id=session_id, message_length=len(message))

    def log_llm_call(
        self, session_id: str, model: str, input_tokens: int, output_tokens: int, latency_ms: float
    ) -> None:
        self._log(
            logging.INFO,
            "llm_call",
            session_id=session_id,
            model=model,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            latency_ms=round(latency_ms, 2),
        )

    def log_tool_call(self, session_id: str, tool: str, allowed: bool) -> None:
        self._log(logging.INFO, "tool_call", session_id=session_id, tool=tool, allowed=allowed)

    def log_error(self, session_id: str, error: str) -> None:
        self._log(logging.ERROR, "error", session_id=session_id, error=error)


if __name__ == "__main__":
    logger = StructuredLogger()
    logger.log_request("s-1", "什么是 RAG？")
    logger.log_llm_call("s-1", "gpt-5.5", 500, 200, 1500.0)
```

### 3.1 日志级别

```python
import logging

logging.debug("模型输入输出全文")       # DEBUG：仅开发，生产关闭
logging.info("请求完成 session=s-1")     # INFO：常规事件
logging.warning("LLM 响应慢，耗时 5.2s")  # WARNING：需关注
logging.error("LLM 调用失败: timeout")   # ERROR：功能受损
logging.critical("向量库不可用")         # CRITICAL：服务不可用
```

> ✅ **结构化优先**：`logger.info("llm_call", model=..., tokens=...)` 可被检索、可聚合；纯文本 `"调用 gpt-5.5，500 tokens"` 难以解析。

## 四、指标

### 4.1 Prometheus 指标（对齐 GenAI 语义）

```python
"""Prometheus 指标：命名对齐 OTel GenAI 语义约定"""
from __future__ import annotations

from fastapi import Response
from prometheus_client import Counter, Gauge, Histogram, generate_latest

# 说明：OTel 属性用点号（gen_ai.request.model），Prometheus 标签名用下划线，转换时自动映射。
LLM_CALLS = Counter(
    "agent_llm_calls_total", "LLM 调用次数",
    ["gen_ai_system", "gen_ai_request_model", "status"],
)
LLM_TOKENS = Counter(
    "agent_llm_tokens_total", "Token 用量",
    ["gen_ai_request_model", "gen_ai_token_type"],  # input / output
)
LLM_LATENCY = Histogram(
    "agent_llm_latency_seconds", "LLM 调用延迟（秒）",
    ["gen_ai_request_model"],
    buckets=[0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0],
)
TOOL_CALLS = Counter(
    "agent_tool_calls_total", "工具调用次数",
    ["tool_name", "status"],
)
ACTIVE_REQUESTS = Gauge("agent_active_requests", "处理中的请求数")
REQUEST_LATENCY = Histogram(
    "agent_request_latency_seconds", "HTTP 请求延迟（秒）",
    ["path", "method"],
    buckets=[0.05, 0.1, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0],
)
REQUEST_ERRORS = Counter(
    "agent_request_errors_total", "请求错误数", ["error_type"],
)


def observe_llm_call(
    model: str, input_tokens: int, output_tokens: int, latency_s: float, ok: bool = True
) -> None:
    """统一记录一次 LLM 调用的指标"""
    status = "ok" if ok else "error"
    LLM_CALLS.labels(gen_ai_system="openai", gen_ai_request_model=model, status=status).inc()
    LLM_TOKENS.labels(gen_ai_request_model=model, gen_ai_token_type="input").inc(input_tokens)
    LLM_TOKENS.labels(gen_ai_request_model=model, gen_ai_token_type="output").inc(output_tokens)
    LLM_LATENCY.labels(gen_ai_request_model=model).observe(latency_s)


def metrics_endpoint() -> Response:
    """暴露 /metrics 供 Prometheus 抓取"""
    return Response(content=generate_latest(), media_type="text/plain")
```

### 4.2 在 FastAPI 中自动采集

```python
"""监控中间件：自动采集请求级指标"""
from __future__ import annotations

import time

from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request


class MetricsMiddleware(BaseHTTPMiddleware):
    """采集 in-flight、请求延迟与错误率"""

    async def dispatch(self, request: Request, call_next):
        ACTIVE_REQUESTS.inc()
        start = time.perf_counter()
        try:
            response = await call_next(request)
            return response
        except Exception:
            REQUEST_ERRORS.labels(error_type="unhandled").inc()
            raise
        finally:
            ACTIVE_REQUESTS.dec()
            REQUEST_LATENCY.labels(
                path=request.url.path, method=request.method
            ).observe(time.perf_counter() - start)
```

> 💡 **核心指标清单**：QPS、P50/P95/P99 延迟、错误率、Token 用量、单位任务成本、工具调用分布、并发数。按 **Agent / 工具 / 模型** 维度拆分。

## 五、链路追踪：OpenTelemetry GenAI 语义约定

### 5.1 统一 span 语义

```python
"""OTel 链路追踪：为 LLM/Agent 打上 gen_ai.* 属性"""
from __future__ import annotations

from opentelemetry import trace

tracer = trace.get_tracer("agentlearn")


def traced_chat(question: str) -> str:
    """给一次 Agent 调用打上 GenAI 语义 span"""
    with tracer.start_as_current_span("gen_ai.chat") as span:
        span.set_attribute("gen_ai.operation.name", "chat")
        span.set_attribute("gen_ai.system", "openai")
        span.set_attribute("gen_ai.request.model", "gpt-5.5")
        span.set_attribute("gen_ai.request.temperature", 0.3)

        # ... 执行 Agent（检索 / 工具 / 多轮 LLM）...

        span.set_attribute("gen_ai.usage.input_tokens", 500)
        span.set_attribute("gen_ai.usage.output_tokens", 200)
        span.set_attribute("gen_ai.response.finish_reasons", ["stop"])
        return "回答内容"
```

### 5.2 导出到后端（OTLP）

```python
"""初始化 OTel：导出到 Langfuse / Jaeger / Grafana Tempo 等 OTLP 后端"""
from opentelemetry.exporter.otlp.proto.http.trace_exporter import OTLPSpanExporter
from opentelemetry.sdk.resources import Resource
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor


def setup_tracing(endpoint: str = "http://localhost:4318/v1/traces") -> TracerProvider:
    """配置全局 TracerProvider"""
    provider = TracerProvider(resource=Resource.create({"service.name": "agent-api"}))
    provider.add_span_processor(BatchSpanProcessor(OTLPSpanExporter(endpoint=endpoint)))
    trace.set_tracer_provider(provider)
    return provider
```

### 5.3 GenAI 常用属性速查

| 属性 | 含义 |
|------|------|
| `gen_ai.system` | 供应方（openai / anthropic / …） |
| `gen_ai.operation.name` | 操作类型（chat / embeddings / execute_tool） |
| `gen_ai.request.model` | 请求模型 |
| `gen_ai.request.temperature` / `max_tokens` | 请求参数 |
| `gen_ai.usage.input_tokens` / `output_tokens` | Token 用量 |
| `gen_ai.response.finish_reasons` | 结束原因 |
| `gen_ai.tool.name` | 工具名（工具调用 span） |

> 💡 主流框架（LangChain / LangGraph）可通过 instrumentation 自动生成 `gen_ai.*` span，无需手写；手写 span 主要用于自定义业务阶段。

## 六、告警

```python
"""告警：阈值判断，超限即通知（可对接企业微信 / 钉钉 / PagerDuty）"""
from __future__ import annotations

from dataclasses import dataclass


@dataclass
class AlertRule:
    """一条告警规则"""

    name: str
    threshold: float
    level: str  # warning / critical


class AlertManager:
    """基于规则的告警判断"""

    def __init__(self) -> None:
        self.rules = {
            "error_rate": AlertRule("error_rate", 0.05, "critical"),
            "p95_latency_s": AlertRule("p95_latency_s", 5.0, "warning"),
            "cost_per_hour_usd": AlertRule("cost_per_hour_usd", 10.0, "warning"),
            "llm_calls_per_request": AlertRule("llm_calls_per_request", 15.0, "warning"),
        }

    def check(self, metrics: dict[str, float]) -> list[dict]:
        alerts: list[dict] = []
        for key, rule in self.rules.items():
            value = metrics.get(key)
            if value is not None and value > rule.threshold:
                alerts.append(
                    {
                        "level": rule.level,
                        "metric": key,
                        "value": value,
                        "threshold": rule.threshold,
                        "message": f"{key}={value} 超过阈值 {rule.threshold}",
                    }
                )
        return alerts
```

配套的 Prometheus 告警规则（`alerts.yml`）：

```yaml
groups:
  - name: agent-api
    rules:
      - alert: AgentHighErrorRate
        expr: sum(rate(agent_request_errors_total[5m])) / sum(rate(agent_llm_calls_total[5m])) > 0.05
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "Agent 错误率超过 5%"
      - alert: AgentHighLatency
        expr: histogram_quantile(0.95, sum(rate(agent_llm_latency_seconds_bucket[5m])) by (le)) > 5
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Agent P95 延迟超过 5s"
```

## 七、三条必看指标

1. **每次请求的 LLM / 工具调用次数**——异常增长往往是**死循环**的前兆（配合 `ModelCallLimitMiddleware` 兜底）；
2. **P95 延迟与失败率**——按 Agent / 工具维度拆分，才能定位瓶颈；
3. **单位任务成本**（token × 单价）——并设置阈值告警（见 [6.3 成本控制](../06-高级主题/03-成本控制.md)）。

## 八、本章小结

| 组件 | 工具 / 方法 | 说明 |
|------|-------------|------|
| **日志** | `JsonFormatter` + `StructuredLogger` | JSON 结构化，便于检索 |
| **指标** | Prometheus（`gen_ai` 命名） | 延迟、QPS、Token、成本可视化 |
| **追踪** | OTel GenAI（`gen_ai.*`） | 全链路 span，定位黑盒 |
| **告警** | 规则 + Prometheus alerting | 错误率 / 延迟 / 成本三线告警 |
| **端点** | `/health` + `/metrics` | 对接探针与抓取 |
| **定义顺序** | `JsonFormatter` 先于 `StructuredLogger` | 避免 `NameError` |
